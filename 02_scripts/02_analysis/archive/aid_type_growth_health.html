attach_iv <- function(sector_data, fin_data, key_cols) {
    sector_data %>%
        left_join(
            fin_data %>%
                select(all_of(c(key_cols, "IV_lag"))) %>%
                distinct(),
            by = key_cols
        )
}

prepare_panel_data <- function(data, admin_level) {
    if (admin_level == "GID_1") {
        data <- data %>%
            arrange(GID_1, study_group, year) %>%
            group_by(GID_1, study_group) %>%
            mutate(
                lag_mean_nl = dplyr::lag(mean_nl),
                nl_growth = case_when(
                    is.na(lag_mean_nl) ~ NA_real_,
                    TRUE ~
                        ((log(mean_nl + 0.01) - log(lag_mean_nl + 0.01)) /
                            log(lag_mean_nl + 0.01)) * 100
                ),
                lag_hhi_admin1 = dplyr::lag(frag_index_admin1),
                lag_pop_admin1 = dplyr::lag(ln_pop_admin1),
                lag_sgq_admin1 = dplyr::lag(mean_sgq_admin1),
                lag_donor_count_admin1 = dplyr::lag(donor_count_admin1),
                lag_total_proj_admin1 = dplyr::lag(total_proj_admin1),
                lag_total_aid_admin1 = log(dplyr::lag(total_aid_admin1) + 0.01)
            ) %>%
            ungroup()
    } else if (admin_level == "GID_2") {
        data <- data %>%
            arrange(GID_2, study_group, year) %>%
            group_by(GID_2, study_group) %>%
            mutate(
                lag_mean_nl = dplyr::lag(mean_nl),
                nl_growth = case_when(
                    is.na(lag_mean_nl) ~ NA_real_,
                    TRUE ~
                        ((log(mean_nl + 0.01) - log(lag_mean_nl + 0.01)) /
                            log(lag_mean_nl + 0.01)) * 100
                ),
                lag_hhi_admin2 = dplyr::lag(frag_index_admin2),
                lag_pop_admin2 = dplyr::lag(ln_pop_admin2),
                lag_sgq_admin2 = dplyr::lag(mean_sgq_admin2),
                lag_donor_count_admin2 = dplyr::lag(donor_count_admin2),
                lag_total_proj_admin2 = dplyr::lag(total_proj_admin2),
                lag_total_aid_admin2 = log(dplyr::lag(total_aid_admin2) + 0.01)
            ) %>%
            ungroup()
    } else {
        stop("admin_level must be either 'GID_1' or 'GID_2'")
    }

    data
}

harmonize_by_admin <- function(data, admin_level) {
    if (admin_level == "GID_1") {
        data <- data %>%
            mutate(
                total_aid = total_aid_admin1,
                frag_index = lag_hhi_admin1,
                lag_log_pop = lag_pop_admin1,
                lag_governance = lag_sgq_admin1,
                lag_donor_count = lag_donor_count_admin1,
                lag_total_proj = lag_total_proj_admin1,
                lag_total_aid = lag_total_aid_admin1
            )
    } else if (admin_level == "GID_2") {
        data <- data %>%
            mutate(
                total_aid = total_aid_admin2,
                frag_index = lag_hhi_admin2,
                lag_log_pop = lag_pop_admin2,
                lag_governance = lag_sgq_admin2,
                lag_donor_count = lag_donor_count_admin2,
                lag_total_proj = lag_total_proj_admin2,
                lag_total_aid = lag_total_aid_admin2
            )
    } else {
        stop("admin_level must be either 'GID_1' or 'GID_2'")
    }

    data
}

mark_governance_groups <- function(data, admin_level, high_cutoff = 0.75) {
    data <- harmonize_by_admin(data, admin_level)
    cutoff <- quantile(data$lag_governance, probs = high_cutoff, na.rm = TRUE)

    data %>%
        mutate(
            high_gov = if_else(
                !is.na(lag_governance) & lag_governance > cutoff,
                1L,
                0L,
                missing = 0L
            ),
            gov_group = if_else(high_gov == 1L, "high", "low")
        )
}

clean_outcome <- function(data, outcome_var) {
    data <- data %>%
        filter(!is.na(.data[[outcome_var]]), is.finite(.data[[outcome_var]]))

    if (outcome_var == "nl_growth" && nrow(data) > 0) {
        growth_cut <- quantile(data[[outcome_var]], probs = c(0.05, 0.95), na.rm = TRUE)
        data <- data %>%
            filter(
                .data[[outcome_var]] > growth_cut[[1]],
                .data[[outcome_var]] < growth_cut[[2]]
            )
    }

    data
}

perform_ols_analysis <- function(data, admin_level, outcome_var, cluster_var) {
    data <- harmonize_by_admin(data, admin_level) %>%
        clean_outcome(outcome_var)

    model_formula <- as.formula(
        paste0(outcome_var, " ~ frag_index + lag_total_aid + lag_log_pop")
    )

    fixest::feols(
        model_formula,
        cluster = as.formula(paste0("~", cluster_var)),
        data = data
    )
}

perform_fe_analysis <- function(data, admin_level, outcome_var, cluster_var) {
    data <- harmonize_by_admin(data, admin_level) %>%
        clean_outcome(outcome_var)

    model_formula <- as.formula(
        paste0(
            outcome_var,
            " ~ frag_index + lag_total_aid + lag_log_pop | GID_0^year + ",
            admin_level
        )
    )

    fixest::feols(
        model_formula,
        cluster = as.formula(paste0("~", cluster_var)),
        data = data
    )
}

perform_cfa_analysis <- function(data, admin_level, outcome_var, iv_var, cluster_var) {
    data <- harmonize_by_admin(data, admin_level) %>%
        clean_outcome(outcome_var)

    stage_1_formula <- as.formula(
        paste0(
            "total_aid ~ ",
            iv_var,
            " + lag_log_pop | GID_0^year + ",
            admin_level
        )
    )

    stage_1 <- fixest::feols(
        stage_1_formula,
        cluster = as.formula(paste0("~", cluster_var)),
        data = data
    )

    data <- data %>%
        mutate(cfa = total_aid - predict(stage_1, newdata = data))

    stage_2_formula <- as.formula(
        paste0(
            outcome_var,
            " ~ frag_index + lag_total_aid + lag_log_pop + cfa | GID_0^year + ",
            admin_level
        )
    )

    fixest::feols(
        stage_2_formula,
        cluster = as.formula(paste0("~", cluster_var)),
        data = data
    )
}

perform_interaction_ols_analysis <- function(data, admin_level, outcome_var, cluster_var) {
    data <- harmonize_by_admin(data, admin_level) %>%
        clean_outcome(outcome_var)

    model_formula <- as.formula(
        paste0(
            outcome_var,
            " ~ frag_index * lag_governance + lag_total_aid + lag_log_pop"
        )
    )

    fixest::feols(
        model_formula,
        cluster = as.formula(paste0("~", cluster_var)),
        data = data
    )
}

perform_interaction_fe_analysis <- function(data, admin_level, outcome_var, cluster_var) {
    data <- harmonize_by_admin(data, admin_level) %>%

    model_formula <- as.formula(
        paste0(

    )

    fixest::feols(

}

run_models_by_aid_type <- function(data, aid_types, model_runner) {
    results <- list()

    for (aid_type in aid_types) {
        aid_data <- data %>% filter(study_group == aid_type)

        fit <- tryCatch(
            model_runner(aid_data),
            error = function(e) {
                message(paste0("Skipped ", aid_type, ": ", e$message))
                NULL
            }
        )

        if (!is.null(fit)) {
            results[[aid_type]] <- fit
        }
    }

    results
}

build_dict <- function(include_cfa = FALSE, include_interaction = FALSE) {
    model_dict <- c(
        frag_index = "Lag Fragmentation",
        lag_total_aid = "Lag LN(Total Aid)",
        lag_log_pop = "Lag LN(Population)"
    )

    if (include_interaction) {
        model_dict <- c(
            model_dict,
            lag_governance = "Lag Governance (SGQ)",
            `frag_index:lag_governance` = "Lag Fragmentation x Lag Governance"
        )
    }

    if (include_cfa) {
        model_dict <- c(model_dict, cfa = "CFA Residual")
    }

    model_dict
}

save_and_show_table <- function(
    model_list,
    file_name,
    include_cfa = FALSE,
    include_interaction = FALSE,
    headers = NULL) {
    if (length(model_list) == 0) {
        return(invisible(NULL))
    }

    if (is.null(headers)) {
        headers <- names(model_list)
    }

    base_args <- c(
        model_list,
        list(
            headers = headers,
            dict = build_dict(include_cfa, include_interaction),
            se.below = TRUE,
            signif.code = c("***" = 0.01, "**" = 0.05, "*" = 0.1),
            fitstat = c("n", "r2")
        )
    )

    do.call(
        fixest::etable,
        c(base_args, list(tex = TRUE, file = here::here("03_output", "tabs", file_name)))
    )

    do.call(fixest::etable, base_args)
}
