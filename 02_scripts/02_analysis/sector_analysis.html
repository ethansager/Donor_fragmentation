# Ensure proper ordering before applying lag()
panel_aid_admin1 <- panel_aid_admin1 %>%
    arrange(GID_1, year) %>%
    group_by(GID_1) %>%
    mutate(
        lag_mean_nl = dplyr::lag(mean_nl),
        nl_growth = case_when(
            is.na(lag_mean_nl) ~ NA_real_,
            TRUE ~
                ((log(mean_nl + 0.01) - log(lag_mean_nl + 0.01)) /
                    log(lag_mean_nl + 0.01)) *
                    100
        ),
        lag_hhi_admin1 = dplyr::lag(frag_index_admin1),
        lag2_hhi_admin1 = dplyr::lag(frag_index_admin1, 2),
        lag3_hhi_admin1 = dplyr::lag(frag_index_admin1, 3),
        lag_pop_admin1 = dplyr::lag(ln_pop_admin1),
        lag_sgq_admin1 = dplyr::lag(mean_sgq_admin1),
        lag_donor_count_admin1 = dplyr::lag(donor_count_admin1),
        lag_total_proj_admin1 = dplyr::lag(total_proj_admin1),
        lag_total_aid_admin1 = log(dplyr::lag(total_aid_admin1) + .01)
    ) %>%
    mutate(
        nl_growth = case_when(
            is.infinite(nl_growth) ~ NA_real_,
            nl_growth > quantile(nl_growth, probs = .05, na.rm = TRUE) ~ NA_real_,
            nl_growth < quantile(nl_growth, probs = .95, na.rm = TRUE) ~ NA_real_,
            TRUE ~ nl_growth
        )
    ) %>%
    ungroup()

prepare_fe_data <- function(
    data,
    admin_level,
    sector_var) {
    if (admin_level == "GID_1") {
        data <- data %>%
            dplyr::mutate(
                total_aid = total_aid_admin1,
                frag_index = lag_hhi_admin1,
                frag_l1 = lag_hhi_admin1,
                frag_l2 = lag2_hhi_admin1,
                frag_l3 = lag3_hhi_admin1,
                lag_log_pop = lag_pop_admin1,
                lag_governance = lag_sgq_admin1,
                lag_donor_count = lag_donor_count_admin1,
                lag_total_proj = lag_total_proj_admin1,
                lag_total_aid = lag_total_aid_admin1
            ) %>%
            filter(study_group == sector_var)
    } else if (admin_level == "GID_2") {
        data <- data %>%
            dplyr::mutate(
                total_aid = total_aid_admin2,
                frag_index = lag_hhi_admin2,
                frag_l1 = lag_hhi_admin2,
                frag_l2 = lag2_hhi_admin2,
                frag_l3 = lag3_hhi_admin2,
                lag_log_pop = lag_pop_admin2,
                lag_governance = lag_sgq_admin2,
                lag_donor_count = lag_donor_count_admin2,
                lag_total_proj = lag_total_proj_admin2,
                lag_total_aid = lag_total_aid_admin2
            ) %>%
            filter(study_group == sector_var)
    }

    data <- data %>%
        dplyr::mutate(
            high_gov = if_else(
                lag_governance > quantile(lag_governance, 0.75, na.rm = TRUE),
                1,
                0
            )
        )

    return(data)
}
perform_stratified_fe_analysis <- function(
    data,
    admin_level,
    outcome_var,
    cluster_var,
    sector_var) {
    data <- prepare_fe_data(data, admin_level, sector_var)

    stage_2_formula <- as.formula(
        paste0(
            outcome_var,
            " ~ lag_total_aid + lag_total_proj + lag_donor_count + lag_log_pop | GID_0^year + ",
            admin_level
        )
    )

    stage_2 <- fixest::feols(
        stage_2_formula,
        cluster = cluster_var,
        data = data
    )

    return(stage_2)
}

run_interaction_fe <- function(
    data,
    admin_level,
    outcome_var,
    cluster_var,
    sector_var,
    gov_spec = c("continuous", "binary")) {
    gov_spec <- match.arg(gov_spec)
    data <- prepare_fe_data(data, admin_level, sector_var)

    rhs_terms <- if (gov_spec == "continuous") {
        "frag_index * lag_governance + lag_total_aid + lag_log_pop"
    } else {
        "frag_index * high_gov + lag_total_aid + lag_log_pop"
    }

    model_formula <- as.formula(
        paste0(
            outcome_var,
            " ~ ",
            rhs_terms,
            " | GID_0^year + ",
            admin_level
        )
    )

    model <- fixest::feols(
        model_formula,
        cluster = cluster_var,
        data = data
    )

    return(model)
}

run_distributed_lag_interaction_fe <- function(
    data,
    admin_level,
    outcome_var,
    cluster_var,
    sector_var,
    gov_spec = c("continuous", "binary")) {
    gov_spec <- match.arg(gov_spec)
    data <- prepare_fe_data(data, admin_level, sector_var)

    rhs_terms <- if (gov_spec == "continuous") {
        paste(
            "frag_l1 + frag_l2 + frag_l3 + lag_governance +",
            "frag_l1:lag_governance + frag_l2:lag_governance +",
            "frag_l3:lag_governance + lag_total_aid + lag_log_pop"
        )
    } else {
        paste(
            "frag_l1 + frag_l2 + frag_l3 + high_gov +",
            "frag_l1:high_gov + frag_l2:high_gov + frag_l3:high_gov +",
            "lag_total_aid + lag_log_pop"
        )
    }

    model_formula <- as.formula(
        paste0(
            outcome_var,
            " ~ ",
            rhs_terms,
            " | GID_0^year + ",
            admin_level
        )
    )

    model <- fixest::feols(
        model_formula,
        cluster = cluster_var,
        data = data
    )

    return(model)
}
